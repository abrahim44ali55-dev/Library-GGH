Full Guide & Implementation

Complete script architecture demonstrating how to combine window creation and interactive elements seamlessly.

local Library = loadstring(game:HttpGet("رابط_السكربت_الخاص_بك_هنا"))()

local Window = Library:CreateWindow({
    Title = "Full Documentation Example",
    Theme = "Dark"
})

local MainTab = Window:CreateTab("General")

MainTab:CreateButton({
    Name = "Execute Action",
    Callback = function()
        print("Action executed successfully!")
    end
})

MainTab:CreateToggle({
    Name = "Enable Loop",
    Default = false,
    Callback = function(State)
        if State then
            print("Feature Enabled")
        else
            print("Feature Disabled")
        end
    end
})
