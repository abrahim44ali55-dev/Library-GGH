1. Loadstring Initialization

Fetch and initialize the UI library securely using your repository raw link.

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/abrahim44ali55-dev/Library-GGH/refs/heads/main/BEST/2026/Nice"))()

2. Window Setup

Initialize the main graphical interface window with core properties.

local Window = Library:CreateWindow({
    Title = "Advanced UI Hub",
    Theme = "Dark",
    ToggleKey = Enum.KeyCode.RightControl
})

3. Themes & Visual Configurations

Switch between available visual styles and color themes for your interface layout dynamically.

Dark Theme Setup:

Library:SetTheme("Dark")

Light Theme Setup:

Library:SetTheme("Light")

Midnight Blue Theme Setup:

Library:SetTheme("Midnight")

Custom Accent Color Configuration:

Library:SetAccent(Color3.fromRGB(88, 166, 255))

4. Complete Core UI Elements (Original Syntax)

Standard base syntax for creating individual components without complex logic.

Tab Creation:

local Tab = Window:CreateTab("Tab Name")

Section Creation:

local Section = Tab:CreateSection("Section Name")

Button Element:

Tab:CreateButton({
    Name = "Button Name",
    Callback = function()
        -- Code execution here
    end
})

Toggle Element:

Tab:CreateToggle({
    Name = "Toggle Name",
    Default = false,
    Callback = function(State)
        -- State returns true or false
    end
})

Slider Element:

Tab:CreateSlider({
    Name = "Slider Name",
    Min = 0,
    Max = 100,
    Default = 50,
    Callback = function(Value)
        -- Value returns number
    end
})

Dropdown Element:

Tab:CreateDropdown({
    Name = "Dropdown Name",
    Options = {"Option 1", "Option 2", "Option 3"},
    Default = "Option 1",
    Callback = function(Selected)
        -- Selected option string
    end
})

Textbox Element:

Tab:CreateTextbox({
    Name = "Textbox Name",
    PlaceholderText = "Type here...",
    Callback = function(Text)
        -- Text string input
    end
})

Label Element:

local Label = Tab:CreateLabel("Information Text Here")

Example 2: Complete Script with Teleports & Notifications

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/abrahim44ali55-dev/Library-GGH/refs/heads/main/BEST/2026/Nice"))()

local Window = Library:CreateWindow({
    Title = "Teleport Hub",
    Theme = "Midnight",
    ToggleKey = Enum.KeyCode.RightShift
})

local Tab = Window:CreateTab("Teleports")

Tab:CreateButton({
    Name = "Teleport to Spawn",
    Callback = function()
        local char = game.Players.LocalPlayer.Character
        if char and char:FindFirstChild("HumanoidRootPart") then
            char.HumanoidRootPart.CFrame = CFrame.new(0, 50, 0)
        end
    end
})

Tab:CreateDropdown({
    Name = "Select Map Location",
    Options = {"Baseplate", "Arena", "Shop"},
    Default = "Baseplate",
    Callback = function(Option)
        print("Selected location: " .. Option)
    end
})

Example 3: Complete Script with Infinite Jump Toggle

local Library = loadstring(game:HttpGet("https://raw.githubusercontent.com/abrahim44ali55-dev/Library-GGH/refs/heads/main/BEST/2026/Nice"))()

local Window = Library:CreateWindow({
    Title = "Utility Hub",
    Theme = "Dark",
    ToggleKey = Enum.KeyCode.Insert
})

local Tab = Window:CreateTab("Misc")

Tab:CreateToggle({
    Name = "Infinite Jump",
    Default = false,
    Callback = function(State)
        _G.InfJump = State
        game:GetService("UserInputService").JumpRequest:Connect(function()
            if _G.InfJump then
                local char = game.Players.LocalPlayer.Character
                if char then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum then
                        hum:ChangeState("Jumping")
                    end
                end
            end
        end)
    end
})
